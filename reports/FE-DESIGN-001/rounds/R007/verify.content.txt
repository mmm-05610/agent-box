# Models FE-CE-016 (head-loss on spawned resource) and FE-CE-017
# (cursor domain invalidation across ns regeneration). MODEL ONLY.

class Stream:
    def __init__(self):
        self.next_seq = 1
    def pump(self):
        s = self.next_seq; self.next_seq += 1; return s
    def resolve(self): return self.next_seq
    def deliver(self, c): return [s for s in range(1, self.next_seq) if s >= c]

# FE-CE-016: adapter pumps inside invoke before subscriber resolves
job = Stream()
job.pump()                    # seq=1, "starting", inside invoke
result = "spawned(job-1)"    # invoke returns
from_cursor = job.resolve()   # subscriber calls cursor.resolve → 2
delivered = job.deliver(from_cursor)
print(f"CE-016: from_cursor={from_cursor} delivered={delivered}")
print(f"CE-016: seq=1 lost? {1 not in delivered}")  # True

# Without pump-before-return
job2 = Stream()
fc2 = job2.resolve()          # 1 (nothing pumped yet)
job2.pump()
print(f"CE-016-OK: from_cursor={fc2} delivered={job2.deliver(fc2)}")

# FE-CE-017: persisted cursor across ns regeneration

for _ in range(20): ns_A.pump()

# Full reconnect → ns_B

delivered = ns_B.deliver(fc)
