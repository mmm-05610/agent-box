EXPERIMENT: does an append-only seq-ordered snapshot log with no invalidate/current marker, re-read from cursor 0 on return, expose "current value" as a core-addressable fact, or only as a per-view derivation from max-seq? This models only the disputed read mechanism; it verifies nothing about the product.

```python
# Model of A's event-plane for a SNAPSHOT resource (git.diff). Proves the model only.
# Question (CE-001): can the core mark a superseded value as not-current?
from dataclasses import dataclass, field

@dataclass
class Env:                      # A.2 Envelope, host-guaranteed order only
    resource_id: str
    seq: int
    kind: str
    payload: str                # the diff text

class CoreStream:               # A.2: content = ordered replayable log, nothing else
    def __init__(self, rid):
        self.rid = rid
        self.log = []
    def ingest(self, payload):  # host assigns monotonic seq on ingest
        s = (self.log[-1].seq + 1) if self.log else 1
        self.log.append(Env(self.rid, s, "git.diff", payload))
    def subscribe(self, from_cursor=0):   # A.2/A.5-S03: replay in seq order
        return [e for e in self.log if e.seq > from_cursor]
    # NO read_value(); NO invalidate(); NO supersede flag on Env -> that is the disputed point.

def main():
    st = CoreStream("git.diff")
    st.ingest("DIFF-V1")        # working tree diff at t0
    st.ingest("DIFF-V2")        # tree changed -> new diff at t1
    returned = st.subscribe(from_cursor=0)   # user leaves page, returns

    # What the core returns on a return-later read:
    seqs = [e.seq for e in returned]
    payloads = [e.payload for e in returned]
    # Is "which entry is current" a CORE fact?
    has_current_flag = any(hasattr(e, "is_current") for e in returned)
    has_superseded_flag = any(hasattr(e, "superseded") for e in returned)
    # Core-addressable current value requires an op; A has none. Derive it the only way:
    derived_current = max(returned, key=lambda e: e.seq).payload

    print("delivered:", seqs, payloads)
    print("core exposes an is_current/superseded flag:", has_current_flag or has_superseded_flag)
    print("core has a read-value op returning ONLY current value: False")
    print("max-seq must be derived by the view to get current diff:", derived_current)

    # Assert the structural claim CE-001 makes:
    assert 1 in seqs and 2 in seqs, "superseded DIFF-V1 is still delivered on return"
    assert not (has_current_flag or has_superseded_flag), "core cannot mark it superseded"
    assert derived_current == "DIFF-V2", "current value only reconstructable from max-seq, not a core fact"
    # A history/timeline resolver that renders all delivered envelopes therefore shows
    # DIFF-V1 alongside DIFF-V2 with no core void marker. That is structural, not a bug.
    print("MODEL RESULT: with stream-as-sole-content, 'current value' is NOT a core primitive;")
    print("superseded snapshots are indistinguishable from live ones at the core layer.")

main()
```
